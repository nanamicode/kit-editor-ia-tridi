// ---------- HUD ----------
const HUD: React.FC = () => {
  const { t, frame, fps } = useT();
  const fase = [...FASES].reverse().find((f) => t >= f.de) ?? FASES[0];

};

// ---------- Tela do jogo (video original) ----------
const blocoPixel = (t: number) => {
  let b = 1;
  for (const T of TRANSICOES) {
    const d = Math.abs(t - T);
    if (d < 0.2) {
      const k = 1 - d / 0.2; // 0..1
      const n = Math.floor(k * 4); // degraus
      b = Math.max(b, [1, 8, 18, 32, 32][n]);
    }
  }
  return b;
};

const palavrasEntre = (de: number, ate: number) => PALAVRAS.filter((p) => p.s >= de - 0.01 && p.s < ate);
const ditoEm = (palavra: string, depoisDe: number) =>
  PALAVRAS.find((p) => p.w.toLowerCase().replace(/[^a-zà-ú0-9]/g, "") === palavra && p.s >= depoisDe - 0.01)?.s ?? 0;

// Entrada em degraus: a caixa "se desenha" de cima para baixo em 4 passos.
const useEntrada = (de: number) => {
  const { frame, fps } = useT();
  const local = frame - Math.round(de * fps);
  const passoN = Math.min(4, Math.max(0, Math.floor(local / 2) + 1));
  return { local, clip: `inset(0 0 ${100 - passoN * 25}% 0)` };
};

const pop = (local: number) => {
  // escala em degraus 1.3 -> 1.15 -> 1
  if (local < 0) return 0;
  if (local < 2) return 1.3;
  if (local < 4) return 1.12;
  return 1;
};

const Legenda: React.FC<{ de: number; ate: number; destaque: string[]; tamanho?: number }> = ({ de, ate, destaque, tamanho = 84 }) => {
  const { t, frame } = useT();
  const ws = palavrasEntre(de, ate);
  const fim = ws.length ? ws[ws.length - 1].e : de;
  const acabou = t >= fim;
  return (
    <div style={{ position: "absolute", left: 40, right: 40, top: 76, bottom: 26 }}>
      <div
        style={{
          fontFamily: vt,
          fontSize: tamanho,
          lineHeight: 1.0,
          color: COR.branco,
          display: "flex",
          flexWrap: "wrap",
          columnGap: "0.4em",
          textShadow: `4px 4px 0 ${COR.tinta}`,
        }}
      >
        {ws.map((p, i) => {
          const visivel = t >= p.s;
          const falando = t >= p.s && t < p.e + 0.05;
          const chave = destaque.includes(p.w.toLowerCase().replace(/[^a-zà-ú0-9]/g, ""));
          return (
            <span
              key={i}
              style={{
                opacity: visivel ? 1 : 0,
                color: falando || chave ? COR.lilas : COR.branco,
              }}
            >
              {p.w}
            </span>
          );
        })}
      </div>
      {acabou && Math.floor(frame / 8) % 2 === 0 ? (
        <div style={{ position: "absolute", right: 0, bottom: 0 }}>
          <Sprite mapa={["ooooooo", ".ooooo.", "..ooo..", "...o..."]} px={7} pal={{ o: COR.lilas }} />
        </div>
      ) : null}
    </div>
  );
};

const CaixaDialogo: React.FC = () => {
  const { t } = useT();
  const atual = MODOS.find((m) => t >= m.de && t < m.ate) ?? MODOS[MODOS.length - 1];
  const { clip, local } = useEntrada(atual.de);
  const rotulo = ROTULO[atual.modo];
  let conteudo: React.ReactNode = null;
  if (atual.modo === "item") conteudo = <ModoItem />;
  if (atual.modo === "fala1") conteudo = <Legenda de={2.0} ate={4.5} destaque={["logo", "embalagem"]} />;
  if (atual.modo === "custo") conteudo = <ModoCusto />;
  if (atual.modo === "inventario") conteudo = <ModoInventario />;
  if (atual.modo === "atributos") conteudo = <ModoAtributos />;
  if (atual.modo === "fala2") conteudo = <Legenda de={12.95} ate={15.05} destaque={["marca"]} />;
  if (atual.modo === "continue") conteudo = <ModoContinue />;
  const rotuloAtivo = local < 6 && Math.floor(local / 2) % 2 === 0;
  return (
    <>
      <MolduraPixel x={CAIXA.x} y={CAIXA.y} w={CAIXA.w} h={CAIXA.h} s={10} borda={8} corBorda={COR.branco} corMiolo="#160830" sombra={14}>
        <div style={{ position: "absolute", inset: 6, border: `4px solid ${COR.roxo}` }} />
        <div style={{ position: "absolute", inset: 0, clipPath: clip }}>{conteudo}</div>
      </MolduraPixel>
      {/* plaquinha com o nome (estilo RPG) */}
      <div
        style={{
          position: "absolute",
          left: CAIXA.x + 40,
          top: CAIXA.y - 30,
          padding: "6px 22px 4px",
          background: rotuloAtivo ? COR.branco : COR.roxo,
          border: `5px solid ${COR.branco}`,
          boxShadow: `6px 6px 0 ${COR.tinta}`,
          fontFamily: silk,
          fontWeight: 400,
          fontSize: 44,
          lineHeight: 1,
          color: rotuloAtivo ? COR.roxo : COR.branco,
        }}
      >
        {rotulo}
      </div>
    </>
  );
};
